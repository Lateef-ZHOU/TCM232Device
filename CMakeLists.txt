cmake_minimum_required(VERSION 3.28)
project(TCM232Device LANGUAGES CXX)

if(NOT TARGET Lidar::HandleCore)
    include(FetchContent)
    set(HANDLECORE_WITH_SERIAL ON)
    FetchContent_Declare(lidar_handle_core
        GIT_REPOSITORY https://github.com/Lateef-ZHOU/HandleCore.git
        GIT_TAG 9cf50f16613a828e0f18bf42285b2e5a67e3a16d)
    FetchContent_MakeAvailable(lidar_handle_core)
endif()
if(NOT TARGET Lidar::Serial)
    message(FATAL_ERROR "Enable HANDLECORE_WITH_SERIAL when configuring HandleCore")
endif()
find_package(Qt6 6.8.3 REQUIRED COMPONENTS Core)
add_library(Tcm232Serial STATIC include/Tcm232Handle.h)
add_library(Lidar::Tcm232Serial ALIAS Tcm232Serial)
set_target_properties(Tcm232Serial PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
target_include_directories(Tcm232Serial PUBLIC include)
target_compile_definitions(Tcm232Serial PUBLIC QT_NO_KEYWORDS)
target_link_libraries(Tcm232Serial PUBLIC Lidar::Serial)
if(MSVC)
    target_compile_options(Tcm232Serial PUBLIC /utf-8)
endif()

if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(serial_delivery tests/serial_delivery.cpp
            "${lidar_handle_core_SOURCE_DIR}/tests/serial/FakeSerialPort.cpp")
        target_include_directories(serial_delivery PRIVATE "${lidar_handle_core_SOURCE_DIR}/tests/serial")
        if(WIN32)
            target_compile_definitions(serial_delivery PRIVATE LIBSERIALPORT_MSBUILD)
        endif()
        target_link_libraries(serial_delivery PRIVATE Lidar::Tcm232Serial)
        add_test(NAME serial_delivery COMMAND serial_delivery)
        set_tests_properties(serial_delivery PROPERTIES TIMEOUT 15
            ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    endif()
endif()
