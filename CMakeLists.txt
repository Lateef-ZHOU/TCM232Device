cmake_minimum_required(VERSION 3.28)
project(TCM232Device LANGUAGES CXX)

if(NOT TARGET Lidar::HandleCore)
    include(FetchContent)
    set(HANDLECORE_WITH_SERIAL ON)
    FetchContent_Declare(lidar_handle_core
        GIT_REPOSITORY https://github.com/Lateef-ZHOU/HandleCore.git
        GIT_TAG 9cf50f16613a828e0f18bf42285b2e5a67e3a16d)
    FetchContent_MakeAvailable(lidar_handle_core)
endif()
if(NOT TARGET Lidar::Serial)
    message(FATAL_ERROR "Enable HANDLECORE_WITH_SERIAL when configuring HandleCore")
endif()
find_package(Qt6 6.8.3 REQUIRED COMPONENTS Core)
add_library(Tcm232Serial STATIC include/Tcm232Handle.h include/Tcm232Packet.h)
add_library(Lidar::Tcm232Serial ALIAS Tcm232Serial)
set_target_properties(Tcm232Serial PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
target_include_directories(Tcm232Serial PUBLIC include)
target_compile_definitions(Tcm232Serial PUBLIC QT_NO_KEYWORDS)
target_link_libraries(Tcm232Serial PUBLIC Lidar::Serial)
if(MSVC)
    target_compile_options(Tcm232Serial PUBLIC /utf-8)
endif()

option(TCM232_WITH_DEVICE "Build the standalone temperature controller backend" ON)
if(TCM232_WITH_DEVICE)
    if(NOT TARGET Lidar::DeviceCore)
        include(FetchContent)
        FetchContent_Declare(lidar_devicecore
            GIT_REPOSITORY https://github.com/Lateef-ZHOU/DeviceCore.git
            GIT_TAG 040ee207bbe9556a1b9b8cbd8117e7c93fbc3001)
        FetchContent_MakeAvailable(lidar_devicecore)
    endif()
    add_library(Tcm232Device STATIC include/Tcm232Device.h)
    add_library(Lidar::Tcm232Device ALIAS Tcm232Device)
    set_target_properties(Tcm232Device PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(Tcm232Device PUBLIC Lidar::Tcm232Serial Lidar::DeviceCore)
endif()

option(TCM232_WITH_QML "Build the temperature controller QML page" ${TCM232_WITH_DEVICE})
if(TCM232_WITH_QML AND TCM232_WITH_DEVICE)
    find_package(Qt6 6.11.2 REQUIRED COMPONENTS Quick Qml)
    if(NOT TARGET Lidar::UICore)
        include(FetchContent)
        FetchContent_Declare(lidar_uicore GIT_REPOSITORY https://github.com/Lateef-ZHOU/UICore.git
            GIT_TAG 9769fdcd28a8a3fdf98f5dde6163e12c6de88563)
        FetchContent_MakeAvailable(lidar_uicore)
    endif()
    get_target_property(ui_dir Lidar::UICore QT_QML_MODULE_OUTPUT_DIRECTORY)
    qt_extract_metatypes(Tcm232Device)
    qt_extract_metatypes(Tcm232Serial)
    set(CMAKE_AUTOMOC ON)
    qt_add_qml_module(Tcm232Types STATIC URI Lidar.Tcm232 VERSION 1.0
        CLASS_NAME Tcm232TypesPlugin RESOURCE_PREFIX /qt/qml
        OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Lidar/Tcm232"
        IMPORT_PATH "${ui_dir}/../.." DEPENDENCIES Lidar.UI
        QML_FILES Tcm232Module.qml
        SOURCES include/Tcm232QmlTypes.h)
    add_library(Lidar::Tcm232Qml ALIAS Tcm232Types)
    target_link_libraries(Tcm232Types PUBLIC Lidar::Tcm232Device Lidar::UICore Qt6::Quick)
endif()

if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(serial_delivery tests/serial_delivery.cpp
            "${lidar_handle_core_SOURCE_DIR}/tests/serial/FakeSerialPort.cpp")
        target_include_directories(serial_delivery PRIVATE "${lidar_handle_core_SOURCE_DIR}/tests/serial")
        if(WIN32)
            target_compile_definitions(serial_delivery PRIVATE LIBSERIALPORT_MSBUILD)
        endif()
        target_link_libraries(serial_delivery PRIVATE Lidar::Tcm232Serial)
        if(TCM232_WITH_DEVICE)
            target_link_libraries(serial_delivery PRIVATE Lidar::Tcm232Device)
            target_compile_definitions(serial_delivery PRIVATE TEST_TCM232_DEVICE)
            target_sources(serial_delivery PRIVATE tests/device.cpp)
            if(WIN32)
                # moc 生成的构造路径也须调用测试替身；生产库仍链接真实串口。
                set_target_properties(serial_delivery PROPERTIES AUTOMOC ON)
                target_sources(serial_delivery PRIVATE include/Tcm232Handle.h include/Tcm232Device.h include/Tcm232Packet.h)
            endif()
        endif()
        if(TARGET Tcm232Types)
            if(WIN32)
                target_compile_definitions(Tcm232Types PRIVATE LIBSERIALPORT_MSBUILD)
            endif()
            target_link_libraries(serial_delivery PRIVATE Tcm232Typesplugin Tcm232Types LidarUICoreplugin)
            target_compile_definitions(serial_delivery PRIVATE TEST_TCM232_QML)
            target_sources(serial_delivery PRIVATE tests/qml.cpp)
        endif()
        add_test(NAME serial_delivery COMMAND serial_delivery)
        set_tests_properties(serial_delivery PROPERTIES TIMEOUT 15
            ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    endif()
endif()
